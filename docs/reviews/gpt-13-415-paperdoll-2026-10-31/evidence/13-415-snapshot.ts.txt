import fs from 'node:fs'; import {config} from 'dotenv'; import {ConvexHttpClient} from 'convex/browser'; import {anyApi} from 'convex/server'; import {kitsFromRegister} from './src/lib/register/stage-kit';
async function main(){config({path:'.env.local',quiet:true});const dir='outputs/catalog-model-repair-2026-10-31/13-415-gpt-paperdoll';fs.mkdirSync(dir,{recursive:true});const c=new ConvexHttpClient(process.env.NEXT_PUBLIC_CONVEX_URL!);
const all:any[]=await c.query(anyApi.products.getAllCatalogGroups,{});const groups=all.filter(g=>g.category==='Glass Bottle'&&g.neckThreadSize==='13-415'&&(g.capacityMl===5||g.applicatorTypes?.includes('Fine Mist Sprayer')));
const details:any[]=[];for(const group of groups)details.push({group,...await c.query(anyApi.products.getProductGroup,{slug:group.slug})});
const products=details.flatMap(d=>d.variants||d.products||[]); console.log('detail keys',Object.keys(details[0]),'product keys',Object.keys(products[0]||{}));
const payload:any={plates:{},components:{},bodies:{},assemblies:{}};for(let i=0;i<products.length;i+=50){const page:any=await c.query(anyApi.registerStage.forSkus,{graceSkus:products.slice(i,i+50).map(p=>p.graceSku)});for(const key of Object.keys(payload))Object.assign(payload[key],page[key]);}
const kits=kitsFromRegister(payload);const overrides=['small-sprayers','cylinder5-profiles','elegant60-reducer','vial-4ml-measured','catalog-followups'];for(const folder of overrides){const p='public/local-kits/'+folder+'-2026-10-31/kits.json';if(fs.existsSync(p)){const local=JSON.parse(fs.readFileSync(p,'utf8')).rows;for(const product of products)if(local[product.websiteSku])kits[product.websiteSku]=local[product.websiteSku];}}
fs.writeFileSync(dir+'/snapshot.json',JSON.stringify({checkedAt:new Date().toISOString(),endpoint:process.env.NEXT_PUBLIC_CONVEX_URL,groups,details,products},null,2));fs.writeFileSync(dir+'/register.json',JSON.stringify(payload));fs.writeFileSync(dir+'/kits-before.json',JSON.stringify(Object.fromEntries(products.filter(p=>kits[p.websiteSku]).map(p=>[p.websiteSku,kits[p.websiteSku]])),null,2));console.log('groups',groups.length,'products',products.length,'kits',products.filter(p=>kits[p.websiteSku]).length);console.log(groups.map(g=>g.slug).join('\n'));
}main().catch(e=>{console.error(e);process.exit(1)});

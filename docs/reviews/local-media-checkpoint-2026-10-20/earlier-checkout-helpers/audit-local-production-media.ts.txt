 const summary={collectedAt:new Date().toISOString(),productionRecords:production.products.length,developmentRecords:development.products.length,missingLocalRecord:rows.filter(r=>!r.localRecordPresent).length,primaryImageDifferences:rows.filter(r=>r.localRecordPresent&&!r.imageSame).length,productionImageButNoLocalImage:rows.filter(r=>r.productionImage&&!r.localImage).length,capOffDifferences:rows.filter(r=>r.localRecordPresent&&!r.capOffSame).length,ringStatus,browserVerified:false};

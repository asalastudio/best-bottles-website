import fs from 'node:fs';import {ConvexHttpClient} from 'convex/browser';import {api} from '../convex/_generated/api';
const file=process.argv[2];
async function main(){const d=JSON.parse(fs.readFileSync(file,'utf8'));const client=new ConvexHttpClient('https://'+d.source);const skus=[...new Set(d.queries.filter((q:any)=>q.name==='products:getProductGroup').flatMap((q:any)=>q.result.variants.map((v:any)=>v.websiteSku)))].filter(Boolean) as string[];const [catalogGroups,primarySkus]=await Promise.all([client.query(api.products.getAllCatalogGroups,{}),client.query(api.products.getCatalogGroupPrimarySkus,{})]);const familyGroups=catalogGroups.filter(g=>g.family===d.family);const ids=new Set(familyGroups.map(g=>g._id));d.catalog={groups:familyGroups,primarySkus:primarySkus.filter(p=>ids.has(p.groupId)),variantPreviewRows:d.previews,capturedAt:new Date().toISOString()};const existing=new Set(d.queries.filter((q:any)=>q.name==='grace:getBottleComponents').map((q:any)=>q.args.websiteSku));let added=0;for(let i=0;i<skus.length;i+=5)await Promise.all(skus.slice(i,i+5).filter(s=>!existing.has(s)).map(async websiteSku=>{const args={websiteSku};const result=await client.query(api.grace.getBottleComponents,args);d.queries.push({name:'grace:getBottleComponents',args,result,capturedAt:new Date().toISOString()});added++;}));if(!fs.existsSync(file+'.initial'))fs.copyFileSync(file,file+'.initial');d.compatibilityEnrichedAt=new Date().toISOString();fs.writeFileSync(file,JSON.stringify(d));fs.copyFileSync(file,'output/local-media-baseline/active.json');console.log({added,queryReceipts:d.queries.length});}main().catch(e=>{console.error(e.message);process.exitCode=1;});
